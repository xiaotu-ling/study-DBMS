     0.先有表:
        USE atguigudb;
        CREATE TABLE IF NOT EXISTS  emp1(
        id INT,
        NAME VARCHAR(25),
        hire_date DATE,
        salary DOUBLE(10,2)
        );
     1.向表中插入新的数据
         SELECT *
         FROM emp1;
