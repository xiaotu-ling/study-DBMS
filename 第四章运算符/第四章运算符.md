         FROM DUAL;

     9.LIKE 模糊查询
         ①查询last_name中包含字符'a'的员工
         SELECT last_name
         FROM employees
         WHERE last_name LIKE =='%a%'==;
         ![[屏幕截图 2026-05-30 225710.png]]
         注意：单引号中间的==%==表示==有不确定字符==，零个，一个或者多个。
         ==如果是'a%'就是查询a开头的==
         ②查询last_name中包含字符'a'，且包含字符'e'的员工信息
         SELECT last_name
         FROM employees
         WHERE last_name LIKE '%a%e%' OR '%e%a%';
         ![[屏幕截图 2026-05-30 230128.png]]
         或者
         SELECT last_name
         FROM employees
         WHERE last_name LIKE '%a%' AND last_name LIKE '%e%';
         ③查询last_name第二个字符是'a'的员工信息
         SELECT last_name
         FROM employees
         WHERE last_name LIKE ' _ a%';
         ==下划线代表有一个字符但不知道是什么==
         ![[屏幕截图 2026-05-30 230537.png]]
         ④查询last_name第二个字符是_且第三个字符是'a'的员工信息

        或者：' _  任意一个其他字符 _a%'  ==ESCAPE== '这个字符’；_
     10.REGEXP，RLIKE正则表达式运算
         '^'来匹配以该字符后面的字符开头的字符串
         '$'来匹配以该字符前面的字符结尾的字符串
         '.'来匹配任何一个单字符
         ‘[]’来匹配在方括号里的任意字符，‘[abc]'匹配“a”或"b"或"c"。'[a-z]'匹配任意字母，'[0-9]'匹配任意数字
         '*'匹配零个或多个在它面前的字符，'x*'匹配任何数量的'x'字符
         SELECT 'abcdef' REGEXP '^a','abcdef' REGEXP 'f$','abcdef' REGEXP 'bc'
         FROM DUAL;

三.逻辑运算符：逻辑运算符返回值为0，1，NULL
     或,且,非,异或，OR || ,AND && ,NOT !,XOR
     异或左右一真一假返回真，例如：
     SELECT last_name,salary,department_id
     FROM employees
     WHERE department_id = 50 XOR salary > 6000;
     查找的是==部门id是50并且工资不大于6000==的员工或者==员工工资大于6000并且部门id不是50==的员工。
